  final String etiqueta;

        children: [

          ),
          Text(
            etiqueta,
            style: const TextStyle(
              color: OL.gris,
              fontSize: 14,
              fontWeight: FontWeight.w700,
              letterSpacing: 0.5,
            ),
          ),
        ],
      ),
    ),
  );
}
